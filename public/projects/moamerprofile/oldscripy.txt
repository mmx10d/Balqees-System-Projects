let original = true;

darkmode.onclick = ()=>{
    if(document.body.style.background == 'black'){
        document.body.style.background = 'white';
        document.body.style.color = 'black';
        darkmode.querySelector('div').style.right = 0;
        setTimeout(()=>{
            darkmode.querySelector('div').innerText = '🌞'
        },100);
    }
    else{
        document.body.style.background = 'black';
        document.body.style.color = 'white';
        darkmode.querySelector('div').style.right = '40px';
        setTimeout(()=>{
            darkmode.querySelector('div').innerText = '🌑'
        },100);
    }
}
lang.onclick = ()=>{

    }else{

        update(AR);
    }
}

function update(language){
    lang.innerText = language.lan;
    document.title = language.page;
    namee.innerText = language.name;
    title.innerText = language.title;
    data.innerHTML = `
        <p>${language.data[0]}</p>
        <p>${language.data[1]}</p>
        <p>${language.data[2]}</p>
        <p>${language.data[3]}</p>
        <p>${language.data[4]}</p>
    `;
    contact.querySelector('.contact_title').querySelector('span').innerText = language.contact.title;
    contact.querySelector('.contact_data').querySelectorAll('a')[0].innerText = language.contact.content[0];
    contact.querySelector('.contact_data').querySelectorAll('a')[1].innerText = language.contact.content[1];
}
